\section{Introducción}


\begin{frame}
  \frametitle{Introducción}
  
  
  Un desarrollo suele ser un trabajo colaborativo entre varias personas. Por ello, hay que resolver varios problemas:
  \begin{itemize}
  \item Cómo dividir el trabajo en varias partes (módulos)
  \item Cómo integrar las partes para conseguir el resultado final.
  \item Cómo comunicar instrucciones para:
    \begin{itemize}
    \item Instalar un entorno de desarrollo (cuando llegan nuevos desarrolladores)
    \item Instalar la aplicación en el servidor de producción
    \item Utilizar los módulos de otras personas
    \item Traspasar el conocimiento (cuando un desarrollador deja el equipo)
    \end{itemize}
  \end{itemize}

\end{frame}

\subsection{Integración de c\'odigo}

\begin{frame}
  \frametitle{Integración de código}
  \begin{itemize}
  \item Cada miembro del equipo realiza su parte del trabajo
  \item Al terminar
    \begin{itemize}
    \item ¿Cómo se integran todas las partes?
    \end{itemize}
  \item Si hay errores o cambios
    \begin{itemize}
    \item ¿Cómo se actualizan solo los nuevos cambios?
    \end{itemize}
  \item Si hay m\'as de una versión
    \begin{itemize}
    \item ¿Cómo se gestionan los conjuntos de módulos compatibles con cierta versión?
    \item ¿Cómo se recupera una versión antigua?
    \item ¿Cómo se fusionan dos versiones?
    \end{itemize}
  \end{itemize}
\end{frame}

\subsection{Control de versiones}

\begin{frame}
  \frametitle{Control de versiones}
  \begin{itemize}
  \item Un control de versiones (de código) mantiene una base de datos con todos los cambios producidos sobre el código fuente y otros ficheros necesarios para generar una aplicación
  \item Pueden ser
    \begin{itemize}
    \item Centralizados/distribuidos
    \item Basados en ficheros/Basados en \textit{changesets}
    \item Bloqueantes/No bloqueantes
    \end{itemize}
  \item Ejemplos
    \begin{itemize}
    \item GIT
    \item Subversion
    \item Mercurial
    \end{itemize}
  \end{itemize}
  \begin{scriptsize}
    \enlace{Control de versiones en Wikipedia}{https://en.wikipedia.org/wiki/Version\_control}
  \end{scriptsize}
\end{frame}


\begin{frame}
  \frametitle{Control de versiones: terminolog\'ia}
  \begin{itemize}
  \item \textbf{Repositorio}: Base de datos con las sucesivas versiones
  \item \textbf{Copia de trabajo}: Ficheros sobre los que trabaja el programador
  \item \textbf{Commit}: Confirmación de una nueva versión (de la copia de trabajo al repositorio)
  \item \textbf{Checkout}: (Re)generación de la copia de trabajo a partir de la información del repositorio
  \item \textbf{Changeset}: Conjunto de cambios en el repositorio, considerado atómico
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Control de versiones: terminolog\'ia}
  \begin{itemize}
  \item \textbf{Rama (Branch)}: Versión paralela (ni anterior, ni posterior)
  \item \textbf{Rama master o trunk}: Rama principal del desarrollo
  \item \textbf{Fusión (Merge)}: Acumular cambios en una misma versión
    \begin{itemize}
    \item Tras un checkout, mezclando la copia de trabajo con la versión del repositorio
    \item Entre diferentes branchs del repositorio
    \end{itemize}
  \item \textbf{Comentario}: Cada commit deber\'ia documentar sus efectos y motivaciones.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{GIT: init}
  \begin{itemize}
  \item Crea un nuevo repositorio en el directorio
  \item Inicialmente vac\'io
  \item Se almacena en el directorio oculto \codigo{.git}

\begin{frame}[fragile]
  \frametitle{GIT: clone}
  \begin{itemize}

  \item El repositorio puede ser local o remoto, dependiendo de la \enlace{https://git-scm.com/docs/git-clone\#URLS}{URL}
  \item La copia de trabajo ser\'a la de la rama MASTER

\begin{frame}
  \frametitle{GIT: add}
  \begin{itemize}
  \item Git no incluye todos los ficheros de la copia de trabajo en el repositorio
  \item Si se desea un fichero en el repositorio, se debe incluir con \texttt{add}
  \item Cada cambio posterior debe indicarse tambi\'en con \texttt{add}
  \item Se puede quitar con \texttt{rm} (las versiones antiguas siguen en el repositorio)

  \vfill

\referenciagit{add}\\
\referenciagit{rm}\\
\referenciagit{mv}\\
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{GIT: status}
  
  \begin{itemize}
  \item Imprime un resumen del estado de los ficheros de la copia de trabajo
    \begin{itemize}
    \item \textbf{Untracked}: No guardados en el repositorio. Git ignora el fichero
    \item \textbf{Unmodified}: Igual que en la rama activa del repositorio
    \item \textbf{Modified}: Con cambios respecto al repositorio, pero que no está previsto guardar
    \item \textbf{Staged}:  El fichero está en el \textit{index}. Será parte del próximo \textit{changeset} que se añadirá al repositorio
    \end{itemize}
  \end{itemize}

  \vfill
\referenciagit{status}
\end{frame}

  \begin{itemize}
  \item Incluye los cambios en el index como una nueva versión en el repositorio
  \item Los ficheros pasan al index con \texttt{add}

  \end{itemize}

\begin{frame}
  \frametitle{GIT: push}
  \begin{itemize}
  \item Sube las versiones del repositorio local a un repositorio remoto
  \item El repositorio remoto puede establecerse:
    \begin{itemize}
    \item con \texttt{git clone} (en este caso se denomina \textbf{origin}) 
    \item con \texttt{git remote add}
    \end{itemize}
  \item El repositorio debe tener todas las versiones del repositorio remoto
    \begin{itemize}
    \item En otro caso, deberá realizarse un \texttt{git pull} previo
    \end{itemize}
  \end{itemize}

  \vfill

\referenciagit{push}\\
\referenciagit{remote}
\end{frame}

\begin{frame}
  \frametitle{GIT: diff}
  Muestra los cambios de los ficheros en estado \textbf{modified}
  \begin{itemize}
  \item \texttt{@@ -lineO,nlinesO +lineD,nlinesD @@}
    \begin{itemize}
    \item Se muestra a partir de \texttt{lineO} del fichero antiguo y \texttt{lineD} del fichero actual
    \item Se muestran \texttt{nlinesO} líneas del fichero antiguo y \texttt{nlinesD} líneas del fichero actual 
    \end{itemize}
  \item \texttt{-}: Línea borrada
  \item \texttt{+}: Línea añadida
  \end{itemize}
  
  \vfill
  \referenciagit{diff}
\end{frame}

\begin{frame}
  \frametitle{GIT: checkout}
  \begin{itemize}
  \item Extrae versiones del repositorio a la copia de trabajo
    \begin{itemize}
    \item \textbf{Un fichero}: \texttt{git checkout \textit{fichero}}
    \item \textbf{Una versión}: \texttt{git checkout \textit{hashdeversión}}
    \item \textbf{Un branch}: \texttt{git checkout \textit{branch}}
    \end{itemize}
  \item También crea nuevas ramas, con \texttt{git checkout -b \textit{nombrerama}}
  \item Si coinciden una versión/nombre de fichero
    \begin{itemize}
    \item Utilizar \enlace{https://www.kernel.org/pub/software/scm/git/docs/gitrevisions.html\#\_specifying\_revisions}{Referencias a objetos en Git}
    \item Es \textbf{mejor} evitar que coincidan
    \end{itemize}

  \end{itemize}

  \vfill
\referenciagit{checkout}\\
\referenciagit{branch}
\end{frame}
